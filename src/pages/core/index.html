// Guardar y cifrar datos confidenciales en localStorage

  storageType: CONSTANT.STORAGE.SOURCE.LOCAL,
  item: "user_secret_data",
  value: { role: "admin", token: "secret_123" },
  secretKey: user.uid // Aislamiento criptográfico por usuario
});

// Descifrar datos
